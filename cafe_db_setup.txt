CREATE TABLE tb_staff (
    staff_id INT         NOT NULL,
    store_id INT         NOT NULL,
    staff_nm VARCHAR(30) NOT NULL,
    role     VARCHAR(20) NOT NULL,   -- MANAGER / BARISTA / PART
    hired_dt DATE        NOT NULL,
    PRIMARY KEY (staff_id),
    FOREIGN KEY (store_id) REFERENCES tb_store(store_id)
) ENGINE=InnoDB COMMENT='직원';

INSERT INTO tb_staff (staff_id, store_id, staff_nm, role, hired_dt)
SELECT seq,
       1 + (seq - 1) DIV 6,
       CONCAT('직원', LPAD(seq, 3, '0')),
       ELT(1 + (seq % 6), 'MANAGER','BARISTA','BARISTA','PART','PART','PART'),
       DATE_ADD('2022-01-01', INTERVAL (seq * 17 % 900) DAY)
FROM seq_1_to_48;

-- ---------------------------------------------------------------------
-- [2] 주문
--     시간대 가중: 08~10시 출근 피크, 12~14시 점심, 15~17시 오후
--     매장 규모(좌석)에 비례해 주문량 배분
-- ---------------------------------------------------------------------
CREATE TABLE tb_order (
    order_id   INT         NOT NULL,
    store_id   INT         NOT NULL,
    staff_id   INT         NOT NULL,
    order_dt   DATETIME    NOT NULL,
    order_type VARCHAR(10) NOT NULL,   -- HALL / TAKEOUT / DELIVERY
    pay_method VARCHAR(10) NOT NULL,   -- CARD / CASH / PAY
    PRIMARY KEY (order_id),
    FOREIGN KEY (store_id) REFERENCES tb_store(store_id),
    FOREIGN KEY (staff_id) REFERENCES tb_staff(staff_id)
) ENGINE=InnoDB COMMENT='주문';

-- [0] 규모 조절: 저사양 PC 는 seq_1_to_120000 → seq_1_to_30000
INSERT INTO tb_order (order_id, store_id, staff_id, order_dt, order_type, pay_method)
SELECT seq,
       st.store_id,
       1 + (st.store_id - 1) * 6 + (seq % 6),
       DATE_ADD(
         DATE_ADD(GREATEST(st.opened_dt, '2023-01-01'),
                  INTERVAL (seq * 31 % GREATEST(DATEDIFF('2024-12-31', GREATEST(st.opened_dt,'2023-01-01')), 1)) DAY),
         INTERVAL (ELT(1 + (seq % 20),
                        7, 8, 8, 9, 9, 9, 10, 10, 11, 12,
                       12, 13, 13, 14, 15, 16, 16, 17, 18, 19) * 60
                   + (seq * 7 % 60)) MINUTE),
       ELT(1 + (seq % 10), 'HALL','HALL','HALL','HALL','TAKEOUT',
                           'TAKEOUT','TAKEOUT','TAKEOUT','DELIVERY','DELIVERY'),
       ELT(1 + (seq % 8), 'CARD','CARD','CARD','CARD','CARD','PAY','PAY','CASH')
FROM seq_1_to_120000 s
JOIN tb_store st ON st.store_id = 1 + (s.seq * 13 % 8);

CREATE TABLE tb_order_item (
    order_item_id INT           NOT NULL AUTO_INCREMENT,
    order_id      INT           NOT NULL,
    menu_id       INT           NOT NULL,
    qty           SMALLINT      NOT NULL,
    unit_price    DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (order_item_id),
    KEY idx_oi_order (order_id),
    FOREIGN KEY (order_id) REFERENCES tb_order(order_id),
    FOREIGN KEY (menu_id)  REFERENCES tb_menu(menu_id)
) ENGINE=InnoDB COMMENT='주문 상세';

INSERT INTO tb_order_item (order_id, menu_id, qty, unit_price)

FROM tb_order o

INSERT INTO tb_order_item (order_id, menu_id, qty, unit_price)
SELECT o.order_id, m.menu_id, 1, m.price

SELECT o.order_id, m.menu_id, 1, m.price
FROM tb_order o
JOIN tb_menu m ON m.menu_id = ELT(1 + (o.order_id % 5), 2,4,24,25,29)

SET SESSION foreign_key_checks = 1;

-- ---------------------------------------------------------------------
-- [3] 통계 갱신 & 확인
-- ---------------------------------------------------------------------
ANALYZE TABLE tb_order, tb_order_item;

SELECT 'tb_city'          AS tbl, COUNT(*) AS cnt FROM tb_city
UNION ALL SELECT 'tb_store',         COUNT(*) FROM tb_store
UNION ALL SELECT 'tb_menu_category', COUNT(*) FROM tb_menu_category
UNION ALL SELECT 'tb_menu',          COUNT(*) FROM tb_menu
UNION ALL SELECT 'tb_staff',         COUNT(*) FROM tb_staff
UNION ALL SELECT 'tb_order',         COUNT(*) FROM tb_order
UNION ALL SELECT 'tb_order_item',    COUNT(*) FROM tb_order_item;

-- 데이터 확인
SELECT st.store_nm,
       COUNT(DISTINCT o.order_id)             AS 주문건수,
       SUM(i.qty * i.unit_price)              AS 매출,
       ROUND(SUM(i.qty * i.unit_price)
             / COUNT(DISTINCT o.order_id))    AS 객단가
FROM tb_order o
JOIN tb_order_item i ON o.order_id = i.order_id
JOIN tb_store st     ON o.store_id = st.store_id
GROUP BY st.store_nm
ORDER BY 매출 DESC;
